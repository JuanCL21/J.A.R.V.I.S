import polars as pl

df = pl.read_csv("ventas_q3.csv")
df_clean = df.drop_nulls(subset=["monto_bruto", "timestamp"])
p95 = df_clean.select(pl.col("monto_bruto").quantile(0.95)).item()

df_clean.write_parquet("metricas.parquet")
print(f"Registros válidos: {len(df_clean)} | Percentil 95: ${p95:,.2f}")
